+class DemoService {
+  private users = [...MOCK_USERS];

+  private settings = { ...MOCK_SETTINGS };
+  private currentUser: UserProfile | null = null;
+  private listeners: Set<() => void> = new Set();
+
+  constructor() {
+    // Load from localStorage if available
+    const saved = localStorage.getItem('messenger_demo_data_v3');
+    if (saved) {
+      const data = JSON.parse(saved);
+      this.users = data.users;
+      this.chats = data.chats;
+      this.messages = data.messages;
+      this.settings = data.settings;
+    } else {
+      // Clear old versions
+      localStorage.removeItem('messenger_demo_data');
+      localStorage.removeItem('messenger_demo_data_v2');
+      this.save();
+    }
+  }
+
+  private save() {
+    localStorage.setItem('messenger_demo_data_v3', JSON.stringify({
+      users: this.users,
+      chats: this.chats,
+      messages: this.messages,
+      settings: this.settings
+    }));
+    this.listeners.forEach(l => l());
+  }
+
+  subscribe(callback: () => void) {
+    this.listeners.add(callback);
+    return () => this.listeners.delete(callback);
+  }
+
+  // Auth
+  login(uid: string) {
+    this.currentUser = this.users.find(u => u.uid === uid) || this.users[0];
+    this.save();
+  }
+
+  logout() {
+    this.currentUser = null;
+    this.save();
+  }
+
+  getCurrentUser() { return this.currentUser; }
+  getUsers() { return this.users; }
+  getChats() { return this.chats.filter(c => c.participants.includes(this.currentUser?.uid || '')); }
+  getSettings() { return this.settings; }
+  
+  getMessages(chatId: string) { return this.messages[chatId] || []; }
+
+  async sendMessage(chatId: string, content: string, type: any = 'text', fileUrl?: string) {
+    if (!this.currentUser) return;
+    
+    const chat = this.chats.find(c => c.id === chatId);
+    if (!chat) return;
+
+    const otherId = chat.participants.find(id => id !== this.currentUser?.uid);
+    if (otherId) {
+      const otherUser = this.users.find(u => u.uid === otherId);
+      if (otherUser && otherUser.blockedUsers.includes(this.currentUser.uid)) {
+        console.warn('Cannot send message: You are blocked by this user.');
+        return;
+      }
+    }
+
+    const msg: Message = {
+      id: Math.random().toString(36).substr(2, 9),
+      chatId,
+      senderId: this.currentUser.uid,
+      content,
+      type,
+      fileUrl,
+      status: 'sent',
+      createdAt: Timestamp.now()
+    };
+    if (!this.messages[chatId]) this.messages[chatId] = [];
+    this.messages[chatId].push(msg);
+    
+    chat.lastMessage = type === 'text' ? content : `[${type}]`;
+    chat.lastMessageAt = Timestamp.now();
+    if (otherId) chat.unreadCount[otherId] = (chat.unreadCount[otherId] || 0) + 1;
+    
+    this.save();
+  }
+

+    const chat = this.chats.find(c => c.id === chatId);
+    if (!chat || !this.currentUser) return;
+    
+    const otherId = chat.participants.find(id => id !== this.currentUser?.uid);
+    if (!otherId) return;
+
+    // Check if current user has blocked this person
+    if (this.currentUser.blockedUsers.includes(otherId)) {
+      console.warn('Blocked user attempted to send a message (mock).');
+      return;
+    }

+    const reply: Message = {
+      id: Math.random().toString(36).substr(2, 9),

+    this.save();
+  }
+
+  updateUserField(uid: string, field: string, value: any) {
+    const idx = this.users.findIndex(u => u.uid === uid);
+    if (idx !== -1) {
+      this.users[idx] = { ...this.users[idx], [field]: value };
+      if (this.currentUser?.uid === uid) this.currentUser = this.users[idx];
+      this.save();
+    }
+  }
+
+  updateSettings(data: Partial<AdminSettings>) {
+    this.settings = { ...this.settings, ...data };
+    this.save();
+  }
+
+  async createChat(otherId: string) {
+    if (!this.currentUser) return;
+    const existing = this.chats.find(c => c.participants.includes(otherId) && c.participants.includes(this.currentUser!.uid));
+    if (existing) return existing.id;
+
+    const newChat: Chat = {
+      id: 'chat-' + Math.random().toString(36).substr(2, 9),
+      participants: [this.currentUser.uid, otherId],

+  }

+  async clearHistory(chatId: string) {

+    const chat = this.chats.find(c => c.id === chatId);

+      chat.lastMessage = '';
+      chat.lastMessageAt = Timestamp.now();
+    }
+    this.save();
+  }

+  async blockUser(targetId: string) {

+    if (isBlocked) {
+      this.currentUser.blockedUsers = this.currentUser.blockedUsers.filter(id => id !== targetId);
+    } else {
+      this.currentUser.blockedUsers.push(targetId);
+    }
+    this.updateUserField(this.currentUser.uid, 'blockedUsers', this.currentUser.blockedUsers);
+  }
+
+  async removeAccess(targetId: string) {
+    if (!this.currentUser || this.currentUser.role !== 'admin') return;
+    const target = this.users.find(u => u.uid === targetId);

+      target.permissions.canChat = false;
+      this.updateUserField(targetId, 'permissions', target.permissions);
+    }
+  }
+
+  async toggleIdentificationBlocked() {
+    if (!this.currentUser) return;
+    this.currentUser.identificationBlocked = !this.currentUser.identificationBlocked;
+    this.updateUserField(this.currentUser.uid, 'identificationBlocked', this.currentUser.identificationBlocked);
+  }
+
+  async toggleMuteUser(targetId: string) {
+    if (!this.currentUser) return;
+    if (!this.currentUser.mutedUsers) this.currentUser.mutedUsers = [];
+    const isMuted = this.currentUser.mutedUsers.includes(targetId);
+    if (isMuted) {
+      this.currentUser.mutedUsers = this.currentUser.mutedUsers.filter(id => id !== targetId);
+    } else {
+      this.currentUser.mutedUsers.push(targetId);
+    }
+    this.updateUserField(this.currentUser.uid, 'mutedUsers', this.currentUser.mutedUsers);
+  }
+
+  async markAsRead(chatId: string, userId: string) {
+    const chat = this.chats.find(c => c.id === chatId);
+    if (chat) {
+      chat.unreadCount[userId] = 0;
+      if (this.messages[chatId]) {
+        this.messages[chatId] = this.messages[chatId].map(m => 
+          m.senderId !== userId ? { ...m, status: 'seen' } : m
+        );
+      }
+      this.save();
+    }
+  }
+
+  async toggleNotificationsBlocked() {
+    if (!this.currentUser) return;
+    this.currentUser.notificationsBlocked = !this.currentUser.notificationsBlocked;
+    this.updateUserField(this.currentUser.uid, 'notificationsBlocked', this.currentUser.notificationsBlocked);
+  }
+
+  async toggleAdminOnly(targetId: string) {
+    if (!this.currentUser || this.currentUser.role !== 'admin') return;
+    const target = this.users.find(u => u.uid === targetId);
+    if (target && target.role === 'prestataire') {
+      target.adminOnly = !target.adminOnly;
+      this.updateUserField(targetId, 'adminOnly', target.adminOnly);
+    }
+  }
+
+  async addReaction(messageId: string, userId: string, emoji: string) {
+    for (const chatId in this.messages) {
+      const msgIdx = this.messages[chatId].findIndex(m => m.id === messageId);
+      if (msgIdx !== -1) {
+        const msg = this.messages[chatId][msgIdx];
+        const reactions = msg.reactions || {};
+        const users = reactions[emoji] || [];
+        
+        if (users.includes(userId)) {
+          reactions[emoji] = users.filter(id => id !== userId);
+          if (reactions[emoji].length === 0) delete reactions[emoji];
+        } else {
+          reactions[emoji] = [...users, userId];
+        }
+        
+        this.messages[chatId][msgIdx] = { ...msg, reactions };
+        this.save();
+        break;
+      }
+    }
+  }
+
+  async toggleIsolation(targetId: string) {
+    if (!this.currentUser || this.currentUser.role !== 'admin') return;
+    const target = this.users.find(u => u.uid === targetId);
+    if (target) {
+      target.isIsolated = !target.isIsolated;
+      this.updateUserField(targetId, 'isIsolated', target.isIsolated);
+    }
+  }
+
+  async toggleRestrictContact(userId: string, targetId: string) {
+    const user = this.users.find(u => u.uid === userId);
+    if (!user) return;
+    
+    if (!user.restrictedContacts) user.restrictedContacts = [];
+    if (!user.exclusionMetadata) user.exclusionMetadata = {};
+    const isRestricted = user.restrictedContacts.includes(targetId);
+    
+    if (isRestricted) {
+      user.restrictedContacts = user.restrictedContacts.filter(id => id !== targetId);
+      delete user.exclusionMetadata[targetId];
+    } else {
+      user.restrictedContacts.push(targetId);
+      user.exclusionMetadata[targetId] = {
+        date: new Date().toLocaleDateString('fr-FR', { day: '2-digit', month: 'long', year: 'numeric' }),
+        reason: 'Isolation de s├®curit├® entre commerciaux du m├¬me secteur.',
+        isPublic: false
+      };
+    }
+    
+    this.updateUserField(userId, 'restrictedContacts', user.restrictedContacts);
+    this.updateUserField(userId, 'exclusionMetadata', user.exclusionMetadata);
+    
+    // Make it mutual for isolation
+    const target = this.users.find(u => u.uid === targetId);
+    if (target) {
+      if (!target.restrictedContacts) target.restrictedContacts = [];
+      if (!target.exclusionMetadata) target.exclusionMetadata = {};
+      if (isRestricted) {
